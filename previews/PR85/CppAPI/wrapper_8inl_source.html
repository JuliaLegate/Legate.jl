145inline Domain domain_from_shape(const Shape& shape) {
146 if (shape.volume() == 0) return Domain::NO_DOMAIN;
147 switch (shape.ndim()) {
148 case 1: {
149 Legion::Point<1> lo = Legion::Point<1>::ZEROES();
150 Legion::Point<1> hi;
151 hi[0] = shape[0] - 1;
152 return Domain(Legion::Rect<1>(lo, hi));
153 }
154 case 2: {
155 Legion::Point<2> lo = Legion::Point<2>::ZEROES();
156 Legion::Point<2> hi;
157 hi[0] = shape[0] - 1;
158 hi[1] = shape[1] - 1;
159 return Domain(Legion::Rect<2>(lo, hi));
160 }
161 case 3: {
162 Legion::Point<3> lo = Legion::Point<3>::ZEROES();
163 Legion::Point<3> hi;
164 hi[0] = shape[0] - 1;
165 hi[1] = shape[1] - 1;
166 hi[2] = shape[2] - 1;
167 return Domain(Legion::Rect<3>(lo, hi));
168 }
169 default:
170 return Domain::NO_DOMAIN;
171 }
172}

370struct GetPtrFunctor {
371 template <legate::Type::Code CODE, int DIM>
372 void* operator()(legate::PhysicalStore* store) {
373#if !LEGATE_DEFINED(LEGATE_USE_CUDA)
374 // Check if FLOAT16 is being used without CUDA support
375 if (CODE == legate::Type::Code::FLOAT16) {
376 throw std::runtime_error(
377 "FLOAT16 type is not supported when building without CUDA");
378 }
379#endif
380 using CppT = typename legate_util::code_to_cxx<CODE>::type;
381 auto shp = store->shape<DIM>();
382 return store->write_accessor<CppT, DIM>().ptr(Realm::Point<DIM>(shp.lo));
383 }
384};
385
392inline void* get_ptr(legate::PhysicalStore* store) {
393 int dim = store->dim();
394 legate::Type::Code code = store->type().code();
395 return legate::double_dispatch(dim, code, GetPtrFunctor{}, store);
396}
