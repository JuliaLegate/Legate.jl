115inline Domain domain_from_shape(const Shape& shape) {
116 if (shape.volume() == 0) return Domain::NO_DOMAIN;
117 switch (shape.ndim()) {
118 case 1: {
119 Legion::Point<1> lo = Legion::Point<1>::ZEROES();
120 Legion::Point<1> hi;
121 hi[0] = shape[0] - 1;
122 return Domain(Legion::Rect<1>(lo, hi));
123 }
124 case 2: {
125 Legion::Point<2> lo = Legion::Point<2>::ZEROES();
126 Legion::Point<2> hi;
127 hi[0] = shape[0] - 1;
128 hi[1] = shape[1] - 1;
129 return Domain(Legion::Rect<2>(lo, hi));
130 }
131 case 3: {
132 Legion::Point<3> lo = Legion::Point<3>::ZEROES();
133 Legion::Point<3> hi;
134 hi[0] = shape[0] - 1;
135 hi[1] = shape[1] - 1;
136 hi[2] = shape[2] - 1;
137 return Domain(Legion::Rect<3>(lo, hi));
138 }
139 default:
140 return Domain::NO_DOMAIN;
141 }
142}

286struct GetPtrFunctor {
287 template <legate::Type::Code CODE, int DIM>
288 void* operator()(legate::PhysicalStore* store) {
289#if !LEGATE_DEFINED(LEGATE_USE_CUDA)
290 // Check if FLOAT16 is being used without CUDA support
291 if (CODE == legate::Type::Code::FLOAT16) {
292 throw std::runtime_error(
293 "FLOAT16 type is not supported when building without CUDA");
294 }
295#endif
296 using CppT = typename legate_util::code_to_cxx<CODE>::type;
297 auto shp = store->shape<DIM>();
298 return store->write_accessor<CppT, DIM>().ptr(Realm::Point<DIM>(shp.lo));
299 }
300};
301
308inline void* get_ptr(legate::PhysicalStore* store) {
309 int dim = store->dim();
310 legate::Type::Code code = store->type().code();
311 return legate::double_dispatch(dim, code, GetPtrFunctor{}, store);
312}
